{$CLEO .cs}
0000: 

while true
wait 0
0A8D: 0@ = read_memory 0xA9AE54 size 4 virtual_protect 1
0A8D: 10@ = read_memory 0x70BDAC size 1 virtual_protect 1
if
0@ == 0 //low
then
    if
    10@ <> 0x85
    then
    0A8C: write_memory 0x70BDAC size 1 value 0x85 virtual_protect 1
    end
else
    if
    10@ <> 0x84
    then
    0A8C: write_memory 0x70BDAC size 1 value 0x84 virtual_protect 1
    end
end
        
end
